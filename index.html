import streamlit as st import pandas as pd import plotly.express as px # ----------------------------------------------------------------------------- # 1. إعدادات الصفحة # ----------------------------------------------------------------------------- st.set_page_config( page_title="داش بورد لوحة متابعة الحوكمة", page_icon="📊", layout="wide", initial_sidebar_state="expanded" ) # تحسين تنسيق الواجهة ولدعم الاتجاه من اليمين لليسار (RTL) st.markdown(""" """, unsafe_allow_html=True) # ----------------------------------------------------------------------------- # 2. ربط وجلب البيانات من Google Sheets تلقائيًا # ----------------------------------------------------------------------------- # استبدل SHEET_ID ورابط الشيت برابط جوجل شيت الخاص بك DEFAULT_SHEET_ID = "YOUR_GOOGLE_SHEET_ID_HERE" st.sidebar.title("⚙️ إعدادات الربط والبيانات") sheet_id = st.sidebar.text_input( "معرّف جوجل شيت (Sheet ID):", value=DEFAULT_SHEET_ID, help="أدخل ID الملف الموجود في رابط Google Sheets" ) # اسم صفحة المصدر الأساسية (المتطلبات الرئيسية) sheet_name = "المتطلبات الرئيسية" @st.cache_data(ttl=60) # تحديث البيانات تلقائيًا كل 60 ثانية def load_data_from_google_sheets(s_id, g_sheet_name): # رابط تصدير الشيت بصيغة CSV تلقائيًا url = f"https://docs.google.com/spreadsheets/d/{s_id}/gviz/tq?tqx=out:csv&sheet={g_sheet_name}" df = pd.read_csv(url) # تنظيف البيانات واختيار الصفوف المكتملة المتطلبات فقط if "الجهة المسؤولة" in df.columns: df = df.dropna(subset=["الجهة المسؤولة"]) # تحويل العمود "الرقم" لنص if "الرقم" in df.columns: df["الرقم"] = df["الرقم"].astype(str).str.replace(".0", "", regex=False) return df try: df_raw = load_data_from_google_sheets(sheet_id, sheet_name) st.sidebar.success("✅ تم الاتصال بجوجل شيت وجلب البيانات بنجاح!") except Exception as e: st.sidebar.error("⚠️ لم يتم الوصول لملف Google Sheets. يرجى التأكد من نشر الملف (Public/Anyone with link) ودقة الـ Sheet ID.") st.stop() # ----------------------------------------------------------------------------- # 3. قائمة الإدارات والوحدات لتصفية البيانات # ----------------------------------------------------------------------------- departments = ["كافة الإدارات (لوحة عامة)"] + sorted(df_raw["الجهة المسؤولة"].dropna().unique().tolist()) st.sidebar.markdown("---") selected_dept = st.sidebar.selectbox("🏢 اختر الإدارة / الوحدة:", departments) st.sidebar.markdown("""
💡 **ملاحظة:** يتم إهمال صفحة "تعليمات الاستخدام" وتعتمد هذه اللوحة كليًا على التحديث الفوري لصفحة المتطلبات الرئيسية. """, unsafe_allow_html=True) # ----------------------------------------------------------------------------- # 4. فلترة البيانات وتقسيمها # ----------------------------------------------------------------------------- if selected_dept == "كافة الإدارات (لوحة عامة)": df_filtered = df_raw.copy() st.title("📊 لوحة متابعة الجاهزية السنوية للحوكمة - العامة") else: df_filtered = df_raw[df_raw["الجهة المسؤولة"] == selected_dept].copy() st.title(f"🏢 لوحة متطلبات جهة: {selected_dept}") # ----------------------------------------------------------------------------- # 5. حساب المؤشرات (KPIs) # ----------------------------------------------------------------------------- total_reqs = len(df_filtered) completed = len(df_filtered[df_filtered["حالة المتطلب"] == "مكتمل"]) in_progress = len(df_filtered[df_filtered["حالة المتطلب"] == "جاري التنفيذ"]) not_started = len(df_filtered[df_filtered["حالة المتطلب"] == "لم يبدأ"]) delayed = len(df_filtered[df_filtered["حالة المتطلب"] == "متأخر"]) not_applicable = len(df_filtered[df_filtered["حالة المتطلب"] == "لا ينطبق"]) # نسبة الإنجاز الإجمالية (تستثني "لا ينطبق" إن وجدت) applicable_total = total_reqs - not_applicable completion_rate = (completed / applicable_total * 100) if applicable_total > 0 else 0.0 # عرض بطاقات المؤشرات col1, col2, col3, col4, col5, col6 = st.columns(6) col1.metric("إجمالي المتطلبات", total_reqs) col2.metric("مكتملة ✅", completed) col3.metric("جاري التنفيذ ⏳", in_progress) col4.metric("لم يبدأ 🔴", not_started) col5.metric("متأخر ⚠️", delayed) col6.metric("نسبة الإنجاز", f"{completion_rate:.1f}%") st.markdown("---") # ----------------------------------------------------------------------------- # 6. الرسم البياني للوحة العامة / أو للجهة # ----------------------------------------------------------------------------- if selected_dept == "كافة الإدارات (لوحة عامة)": st.subheader("📌 تحليل جاهزية الحوكمة حسب الإدارة / الوحدة") # تجميع البيانات حسب الجهة والحالة dept_stats = df_raw.groupby(["الجهة المسؤولة", "حالة المتطلب"]).size().reset_index(name="العدد") color_map = { "مكتمل": "#2ec4b6", "جاري التنفيذ": "#ff9f1c", "لم يبدأ": "#e71d36", "متأخر": "#d62828", "لا ينطبق": "#8d99ae" } fig = px.bar( dept_stats, x="الجهة المسؤولة", y="العدد", color="حالة المتطلب", title="توزيع حالات المتطلبات لكل إدارة ووحدة", color_discrete_map=color_map, barmode="stack", height=500 ) fig.update_layout(xaxis_title="الإدارة / الوحدة", yaxis_title="عدد المتطلبات") st.plotly_chart(fig, use_container_width=True) # ----------------------------------------------------------------------------- # 7. عرض جدول بيانات المتطلبات منفصل ومفصّل # ----------------------------------------------------------------------------- st.subheader("📋 جدول المتطلبات والتحديثات التفصيلية") # خيارات تصفية إضافية داخل الجدول col_search, col_status = st.columns([2, 1]) with col_status: status_filter = st.multiselect( "تصفية حسب الحالة:", options=list(df_raw["حالة المتطلب"].unique()), default=list(df_raw["حالة المتطلب"].unique()) ) with col_search: search_query = st.text_input("🔍 بحث في نص المتطلب التنفيذي أو الأدلة المطلوبة:") # تطبيق الفلاتر df_table = df_filtered[df_filtered["حالة المتطلب"].isin(status_filter)] if search_query: df_table = df_table[ df_table["المتطلب التنفيذي"].astype(str).str.contains(search_query, case=False, na=False) | df_table["الشاهد/الأدلة المطلوبة"].astype(str).str.contains(search_query, case=False, na=False) ] # عرض الأعمدة المهمة فقط display_cols = [ "الرقم", "المتطلب التنفيذي", "الشاهد/الأدلة المطلوبة", "الجهة المسؤولة", "حالة المتطلب", "تاريخ الاستحقاق", "آخر تحديث", "رابط الشاهد", "ملاحظات وحدة الحوكمة والتميز" ] display_cols = [col for col in display_cols if col in df_table.columns] st.dataframe( df_table[display_cols], use_container_width=True, hide_index=True )
