import streamlit as st import pandas as pd import plotly.express as px import urllib.parse # ----------------------------------------------------------------------------- # 1. إعدادات الصفحة والتصميم # ----------------------------------------------------------------------------- st.set_page_config( page_title="داش بورد لوحة متابعة الحوكمة", page_icon="📊", layout="wide", initial_sidebar_state="expanded" ) # دعم اللغة العربية والاتجاه من اليمين لليسار (RTL) st.markdown(""" """, unsafe_allow_html=True) # ----------------------------------------------------------------------------- # 2. إعدادات الربط مع Google Sheets # ----------------------------------------------------------------------------- st.sidebar.title("⚙️ ربط Google Sheets") sheet_id = st.sidebar.text_input( "معرّف ملف جوجل شيت (Sheet ID):", value="1xxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxx", help="أدخل ID الملف الموجود في رابط Google Sheets بين /d/ و /edit" ) # ضمان اختيار صفحة المتطلبات الرئيسية وإهمال التعليمات sheet_name = "المتطلبات الرئيسية" gid = st.sidebar.text_input("معرّف الصفحة (GID) - اختياري:", value="", help="إذا كانت الصفحة الرئيسية ليست الأولى، أدخل رقم GID الخاص بصفحة المتطلبات الرئيسية") @st.cache_data(ttl=30) # تحديث البيانات تلقائياً كل 30 ثانية def load_data_from_google_sheets(s_id, g_name, g_id): encoded_sheet_name = urllib.parse.quote(g_name) # بناء رابط التصدير التلقائي بناءً على GID أو اسم الصفحة if g_id.strip(): url = f"https://docs.google.com/spreadsheets/d/{s_id}/export?format=csv&gid={g_id.strip()}" else: url = f"https://docs.google.com/spreadsheets/d/{s_id}/gviz/tq?tqx=out:csv&sheet={encoded_sheet_name}" df = pd.read_csv(url) # ------------------------------------------------------------------------- # استبعاد صفحة التعليمات وتنظيف البيانات # ------------------------------------------------------------------------- # إذا كانت القراءة سحبت صفحة التعليمات عن الخطأ، سنقوم بتصفية الأعمدة والصفوف if "الجهة المسؤولة" not in df.columns: # البحث عن السطر المبتدئ بالرقم والمتطلب df.columns = df.iloc[0] df = df[1:] # تصفية الصفوف التي تحتوي على بيانات المتطلبات فقط وإهمال الصفوف الفارغة/التعليمات if "الجهة المسؤولة" in df.columns: df = df.dropna(subset=["الجهة المسؤولة"]) df = df[df["الجهة المسؤولة"].astype(str).str.strip() != ""] df = df[~df["الجهة المسؤولة"].astype(str).str.contains("تعليمات|مصدر البيانات", na=False)] # معالجة عمود الرقم وحالة المتطلب if "الرقم" in df.columns: df["الرقم"] = df["الرقم"].astype(str).str.replace(".0", "", regex=False) if "حالة المتطلب" in df.columns: df["حالة المتطلب"] = df["حالة المتطلب"].fillna("لم يبدأ").astype(str).str.strip() return df # جلب البيانات try: df_raw = load_data_from_google_sheets(sheet_id, sheet_name, gid) st.sidebar.success(f"✅ تم الاتصال بنجاح | إجمالي المتطلبات: {len(df_raw)}") except Exception as e: st.sidebar.error("⚠️ فشل الاتصال برابط Google Sheets. يُرجى التأكد من نشر الملف (Publish to web) وتحديد الـ Sheet ID الصحيح.") st.stop() # ----------------------------------------------------------------------------- # 3. تصفية وتقسيم الإدارات والوحدات # ----------------------------------------------------------------------------- st.sidebar.markdown("---") st.sidebar.subheader("🏢 تصفية الإدارة / الوحدة") all_departments = sorted(df_raw["الجهة المسؤولة"].unique().tolist()) dept_options = ["📋 كافة الإدارات (اللوحة العامة)"] + all_departments selected_dept_option = st.sidebar.selectbox("اختر الإدارة / الوحدة:", dept_options) # ----------------------------------------------------------------------------- # 4. معالجة البيانات وبناء الداش بورد # ----------------------------------------------------------------------------- if selected_dept_option == "📋 كافة الإدارات (اللوحة العامة)": df_filtered = df_raw.copy() st.title("📊 لوحة متابعة الحوكمة والجاهزية السنوية - العامة") st.caption("تحديث تلقائي وفوري من Google Sheets (إهمال صفحة التعليمات ومعالجة كافة المتطلبات)") else: selected_dept = selected_dept_option df_filtered = df_raw[df_raw["الجهة المسؤولة"] == selected_dept].copy() st.title(f"🏢 داش بورد: {selected_dept}") st.caption("متابعة المتطلبات والتحديثات الخاصة بالإدارة بشكل منفصل") # حساب أرقام وتفاصيل الإنجاز total_reqs = len(df_filtered) completed = len(df_filtered[df_filtered["حالة المتطلب"] == "مكتمل"]) in_progress = len(df_filtered[df_filtered["حالة المتطلب"] == "جاري التنفيذ"]) not_started = len(df_filtered[df_filtered["حالة المتطلب"] == "لم يبدأ"]) delayed = len(df_filtered[df_filtered["حالة المتطلب"] == "متأخر"]) not_applicable = len(df_filtered[df_filtered["حالة المتطلب"] == "لا ينطبق"]) applicable_total = total_reqs - not_applicable completion_rate = (completed / applicable_total * 100) if applicable_total > 0 else 0.0 # ----------------------------------------------------------------------------- # 5. عرض كروت المؤشرات (KPIs) # ----------------------------------------------------------------------------- kpi1, kpi2, kpi3, kpi4, kpi5, kpi6 = st.columns(6) kpi1.metric("إجمالي المتطلبات", total_reqs) kpi2.metric("مكتملة ✅", completed) kpi3.metric("جاري التنفيذ ⏳", in_progress) kpi4.metric("لم يبدأ 🔴", not_started) kpi5.metric("متأخر ⚠️", delayed) kpi6.metric("نسبة الإنجاز", f"{completion_rate:.1f}%") st.markdown("---") # ----------------------------------------------------------------------------- # 6. الرسوم البيانية التفاعلية # ----------------------------------------------------------------------------- color_discrete_map = { "مكتمل": "#2a9d8f", "جاري التنفيذ": "#e9c46a", "لم يبدأ": "#e76f51", "متأخر": "#d62828", "لا ينطبق": "#a8ded6" } if selected_dept_option == "📋 كافة الإدارات (اللوحة العامة)": st.subheader("📊 تحليل الجاهزية وحالات المتطلبات حسب الجهة") # تجميع الحالات لكل إدارة dept_summary = df_raw.groupby(["الجهة المسؤولة", "حالة المتطلب"]).size().reset_index(name="العدد") fig = px.bar( dept_summary, x="الجهة المسؤولة", y="العدد", color="حالة المتطلب", color_discrete_map=color_discrete_map, barmode="stack", height=450 ) fig.update_layout( xaxis_title="الإدارة / الوحدة", yaxis_title="عدد المتطلبات", font_family="Cairo", legend_title_text="حالة المتطلب" ) st.plotly_chart(fig, use_container_width=True) else: col_chart1, col_chart2 = st.columns([1, 1]) with col_chart1: st.subheader("📌 توزيع حالات المتطلبات للإدارة") status_counts = df_filtered["حالة المتطلب"].value_counts().reset_index() status_counts.columns = ["حالة المتطلب", "العدد"] fig_pie = px.pie( status_counts, values="العدد", names="حالة المتطلب", color="حالة المتطلب", color_discrete_map=color_discrete_map, hole=0.4 ) fig_pie.update_layout(font_family="Cairo") st.plotly_chart(fig_pie, use_container_width=True) with col_chart2: st.subheader("💡 ملخص التقييم السريع") st.info(f""" - **الجهة:** {selected_dept_option} - **المتطلبات المكتملة:** {completed} من أصل {total_reqs} - **المتطلبات قيد العمل:** {in_progress} - **المتطلبات التي لم تبدأ:** {not_started} - **معدل الإنجاز الحالي:** {completion_rate:.1f}% """) # ----------------------------------------------------------------------------- # 7. عرض جدول بيانات المتطلبات التفاعلي # ----------------------------------------------------------------------------- st.markdown("---") st.subheader("📋 جدول متطلبات الحوكمة والتحديثات المباشرة") # تصفية تفاعلية داخل الجدول c_search, c_filter = st.columns([3, 1]) with c_filter: status_select = st.multiselect( "فلترة حسب الحالة:", options=list(df_raw["حالة المتطلب"].unique()), default=list(df_raw["حالة المتطلب"].unique()) ) with c_search: search_txt = st.text_input("🔍 بحث برقم المتطلب، النص، أو الشاهد المطلوبة:") # تطبيق الفلاتر df_view = df_filtered[df_filtered["حالة المتطلب"].isin(status_select)] if search_txt: df_view = df_view[ df_view["المتطلب التنفيذي"].astype(str).str.contains(search_txt, na=False) | df_view["الشاهد/الأدلة المطلوبة"].astype(str).str.contains(search_txt, na=False) | df_view["الرقم"].astype(str).str.contains(search_txt, na=False) ] # استعراض الجدول cols_to_show = [ "الرقم", "المتطلب التنفيذي", "الشاهد/الأدلة المطلوبة", "الجهة المسؤولة", "حالة المتطلب", "تاريخ الاستحقاق", "آخر تحديث", "رابط الشاهد", "ملاحظات وحدة الحوكمة والتميز" ] available_cols = [c for c in cols_to_show if c in df_view.columns] st.dataframe( df_view[available_cols], use_container_width=True, hide_index=True )
